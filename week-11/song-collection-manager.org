* Data Structures/Playlist Search
#+name: playlist-search
#+begin_src cpp :tangle src/playlist-search.cpp
<<pledge>>
#include <iostream>
#include <vector>
#include <string>

int main() {
  // Create a playlist with initial zongs
  std::vector<std::string> playlist {"Rock Anthem", "Jazz Groove", "Pop Hit", "Blues Tune"};

  // use iterators to print in forward order, displaying song and index
  int index = 1;
  std::cout << "Song Collection:" << '\n';
  for (auto it = playlist.begin(); it < playlist.end(); it++) {
    std::cout << index << ": " << *it << '\n';
    index++;
  }

  std::cout << '\n'; // Line break

  const std::string searchCondition { "Jazz Groove" };
  bool resultFound = false;

  // Iterate and over the playlist, mark bool true if found.
  for(auto it = playlist.begin(); it != playlist.end(); it++) {
    if (*it == searchCondition)
      resultFound = true;
  }

  // Pretty print whether it was found or not (based on bool...)
  if (resultFound == true) { std::cout << searchCondition << " is in the collection!" << '\n'; }
  else { std::cout << searchCondition << " not found!" << '\n'; }

  return 0; // Exit normally
 }
#+end_src
