#include <vector>
#include <string>
#include <iostream>

// fn prototype for populating rooms
std::vector<std::string> populateRooms(int&, const int);

int main() {

  int N {}; // Var holds number of classrooms
  int M {}; // Var holds number of People

  // Take input from the Kattis judge app:
  std::cin >> N;
  std::cin >> M;

  // Construct classrooms vector:
  auto Rooms = populateRooms(M, N);

  // Output each room in Rooms as a line of *'s:
  for (std::string room: Rooms) {
    std::cout << room << '\n';
  }

  // Courteously flush the buffer... (good practice)
  std::cout << std::endl;

  // Exit normally
  return 0;
}

std::vector<std::string> populateRooms(int& headCount, const int numRooms) {
  /*
  ,Description: populateRooms function:
  ,takes in a reference to headCount (so it can be changed), and numRooms (expressly not to change)
  ,Returns a vector that has the strings with asterisks * for each student.
  ,*/

  // floor division: how many people can fit evenly in each room?
  int numPeople_evenly = headCount/numRooms;
  // Decrement headCount by total# people who now are evenly divided into rooms:
  headCount -= numPeople_evenly * numRooms;

  // Construct a string with ***... appropriate number of *'s
  // for num people can fit evenly in the rooms:
  std::string numPeople_str(numPeople_evenly, '*');

  // Construct a vector Rooms of N numRooms, default contains numPeopleStr people per room...
  std::vector<std::string> Rooms (numRooms, numPeople_str);

  // Finish populating the odd number of people if needed:
  // NOTE: this should always be safe, since the remainder is always less than numRooms...
  while (headCount > 0) {
    // Careful to consider that rooms starts at 0:
    //   So, this is essentially roomIndex working backwards until there's none left.
    //   at -> reference to STRING... use push_back()
    Rooms.at(headCount-1).push_back('*');

    // Decrement (remaining) headCount
    headCount--;
  }

  return Rooms;
}
