* Solution:
#+begin_src cpp :noeval :tangle src/faktor.cpp
#include <iostream>

int main() {
  // Init and take values from user...
  //   A -> The number of Articles
  //   I -> The desired impact
  int A = 0;
  int I = 0;
  std::cin >> A;
  std::cin >> I;

  float C = A*I; // the number of citations needed, set to product and iterated over

  /*
   I had to invent a way to tell C++ how to do ceiling calculations...
   Since ceil(23.0001) = 24, I'm using I-1 as the starting point.

   So, I'm decreasing C incrementally until I'm just shy of I-1...
   The while loop technically goes one too many, so C+1 is what's printed.
   ,*/
  while (C/A > I-1) {
    --C;
  }

  // Provide the calculated value for C:
  std::cout << C+1 << '\n';

  // Exit Normally...
  return 0;
}
#+end_src
