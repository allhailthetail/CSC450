#ifndef CONTACTINFO_H
#define CONTACTINFO_H
#include <string>

class ContactInfo {
private:
  std::string name;
  std::string phone;
public:
  ContactInfo(const std::string& n, const std::string& p) {
    name  = n;
    phone = p;
  }

  // Setter
  void setPhoneNumber(const std::string& newPhone) { phone = newPhone; }

  // Destructor... Is this even needed? It dies when it goes out of scope...
  // This could be left to use the default, or you can define some custom behavior.
  // Regardless, C++ automagically destroys the strings and hands that memory back to the kernel, so
  // doing this explicitly (while possible) isn't necessary.
  // Actually, I don't think that's the case.  If so, why does the message print twice when the program runs?
  //~ContactInfo() { std::cout << "Contact was destroyed..." << '\n'; }

  const std::string& getName()        const { return name; }
  const std::string& getPhoneNumber() const { return phone; }
};

#endif
