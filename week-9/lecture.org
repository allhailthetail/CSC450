#include <iostream>
#include <string>
#include <sstream>
using namespace std;

class Player {
private:
    string name;
    int hp;
    int ammo;
    int x, y;

    int clamp(int v, int lo, int hi) {
        if (v < lo) return lo;
        if (v > hi) return hi;
        return v;
    }

public:
    Player(string n, int h, int a)
        : name(n), hp(clamp(h,0,100)), ammo(a), x(0), y(0) {}

    Player& move(int dx, int dy) {
        x += dx; y += dy;
        return *this;
    }

    Player& fire(int rounds) {
        ammo -= rounds;
        if (ammo < 0) ammo = 0;
        return *this;
    }

    Player& take_damage(int dmg) {
        hp = clamp(hp - dmg, 0, 100);
        return *this;
    }

    Player& reload(int rounds) {
        ammo += rounds;
        return *this;
    }

    Player& heal(int pts) {
        hp = clamp(hp + pts, 0, 100);
        return *this;
    }

    string status() const {
        ostringstream os;
        os << name << " HP=" << hp
           << "  Ammo=" << ammo
           << " Pos=(" << x << "," << y << ")";
        return os.str();
    }
};

int main() {
    Player p{"Daniels", 100, 30};
    cout << p.status() << "\n";
    p.move(1,0).fire(10).take_damage(25).reload(15).heal(10);
    cout << p.status() << "\n";
}

    // Fixed - needed

    // Fixed - needed
    void fire(int rounds) { if (ammo > 0) ammo -= rounds; } // bug, could fire more than ammo I have? Negative ammo?

    void reload(int rounds) { ammo += rounds; }             // bug here, player could accumulate more ammo than int can hold?

    // Fixed - needed Slightly longer, but cleaner...
    void health(int delta_health) {
        if (hp + delta_health < 0) {
            hp = 0;
            cout << "You died..." << '\n';
        } else if (hp + delta_health > MAX_HP) {
            cout << "Health is now max..." << '\n';
            hp = 100;
        } else { hp += delta_health; }
    }

    // Fixed - needed
    string status() const {
        // Init a string to store status in...
        string status;
        status = name + " HP=" + to_string(hp)
            + " Ammo=" + to_string(ammo)
            + " Pos=(" + to_string(x) + ","
            + to_string(y) + ")";

        return status;  // Return the concatenated string
    }
};

    Player p;  // Just use the default constructor

    // Show initial player status
    cout << p.status() << "\n"; // fixed

    p.fire(10);      // -10 ammo
    p.health(-25);   // minus 25 HP
    p.reload(15);    // +15 ammo

    cout << p.status() << "\n";  // Show player status
